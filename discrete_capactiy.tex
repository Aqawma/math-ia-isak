\section{Discrete Capactity \& the Floor Function}
\subsection{Limits of Calculus}
For a fixed wagon width, $W$, the number of bales of width $w$ that can fit can be described by the function
\[
N(w)=\left\lfloor\frac{W}{w}\right\rfloor
\]
We define $k$ as the number of bales that can fit in a row, or the output of $N$. As such, we can define the domain of $w$ in which $k$ bales can fit in a row as:
\[
\frac{W}{k+1}<w\leq\frac{W}{k}.
\]
There will be a jump at $w=\frac{W}{k}$ for every positive integer $k$, and between the jumps $N'(w)=0$.
\[
\lim_{w\rightarrow(W/k)^-}N(w)=k, \qquad \lim_{w\rightarrow(W/k)^+}N(w)=k-1
\]

\subsection{Smoothing the Step with an Expectation}

\[
\mathbb{E}\left[\left\lfloor\frac{W}{W_b}\right\rfloor\right]=\sum_{k=1}^{\infty}k\mathbb{P}\left(\frac{W}{k+1}<W_b\leq\frac{W}{k}\right)
\]
\[
\left[\left\lfloor\frac{W}{W_b}\right\rfloor\right]=\sum_{k=1}^{\infty}k\left[\Phi\left(\frac{W/k-\mu}{\sigma}\right)-\Phi\left(\frac{W/(k+1)-\mu}{\sigma}\right)\right]
\]

\[
\frac{d}{d\mu}\mathbb{E}\left[\left\lfloor\frac{W}{W_b}\right\rfloor\right]=\frac{1}{\sigma}\sum_{k=1}^{\infty}k\left[\phi\left(\frac{W/(k+1)-\mu}{\sigma}\right)-\phi\left(\frac{W/k-\mu}{\sigma}\right)\right]
\]
